cmake_minimum_required(VERSION 3.20)
project(SpellcrossAssetConverter VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SC_CONVERTER_GUI "Build the desktop application" ON)
add_library(spellcross_converter_core src/core/inventory.cpp src/core/graphics.cpp)
target_include_directories(spellcross_converter_core PUBLIC include)
if(SC_CONVERTER_GUI)
  find_package(wxWidgets REQUIRED COMPONENTS core base)
  include(${wxWidgets_USE_FILE})
  add_executable(SpellcrossConverter WIN32 src/gui/main.cpp)
  target_link_libraries(SpellcrossConverter PRIVATE spellcross_converter_core ${wxWidgets_LIBRARIES})
  install(TARGETS SpellcrossConverter RUNTIME DESTINATION bin)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(converter_tests tests/inventory_tests.cpp)
  target_link_libraries(converter_tests PRIVATE spellcross_converter_core)
  add_test(NAME inventory COMMAND converter_tests)
endif()
