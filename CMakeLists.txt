cmake_minimum_required(VERSION 3.20)
project(SpellcrossAssetConverter VERSION 0.9.14 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SC_CONVERTER_GUI "Build the desktop application" ON)
option(SC_BUNDLE_FFMPEG "Provision standalone FFmpeg for Windows x64 builds" ON)
set(RELOADED_VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/vendor/reloaded")
add_library(sc_smacker SHARED "${RELOADED_VENDOR}/third_party/libsmacker/smacker.c")
set_target_properties(sc_smacker PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
add_library(spellcross_converter_core src/core/inventory.cpp src/core/graphics.cpp src/core/video.cpp src/core/pack.cpp src/core/semantic.cpp src/core/campaign.cpp src/core/music.cpp src/core/validation.cpp src/core/presentation_data.cpp
    "${RELOADED_VENDOR}/spellcross-remaster-main/source/smacker_video_decoder.cpp")
target_include_directories(spellcross_converter_core PRIVATE "${RELOADED_VENDOR}/spellcross-remaster-main/source")
target_link_libraries(spellcross_converter_core PRIVATE sc_smacker)
target_include_directories(spellcross_converter_core PUBLIC include)
add_executable(sc_xmidi_worker src/tools/xmidi_worker.cpp)
target_include_directories(sc_xmidi_worker PRIVATE "${RELOADED_VENDOR}/spellcross-remaster-main/source/cxxmidi/guts")
target_link_libraries(sc_xmidi_worker PRIVATE spellcross_converter_core)
if(SC_CONVERTER_GUI)
  find_package(Threads REQUIRED)
  find_package(wxWidgets REQUIRED COMPONENTS core base adv)
  include(${wxWidgets_USE_FILE})
  add_executable(SpellcrossConverter WIN32 src/gui/main.cpp)
  target_link_libraries(SpellcrossConverter PRIVATE spellcross_converter_core ${wxWidgets_LIBRARIES} Threads::Threads)
  if(WIN32 AND SC_BUNDLE_FFMPEG AND CMAKE_SIZEOF_VOID_P EQUAL 8 AND
      NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^[Aa][Rr][Mm]|^[Aa][Aa][Rr][Cc][Hh]")
    include(cmake/BundleFFmpeg.cmake)
  endif()
  if(UNIX AND NOT APPLE)
    set_target_properties(SpellcrossConverter PROPERTIES INSTALL_RPATH "$ORIGIN/../lib")
  endif()
  install(TARGETS SpellcrossConverter sc_smacker sc_xmidi_worker RUNTIME DESTINATION bin LIBRARY DESTINATION lib)
  install(FILES vendor/reloaded/LICENSE DESTINATION share/spellcross-converter RENAME LICENSE-CPAL.txt)
  install(FILES vendor/reloaded/third_party/libsmacker/COPYING DESTINATION share/spellcross-converter RENAME LICENSE-libsmacker.txt)
  install(FILES vendor/reloaded/docs/legal/LICENSES/LGPL-2.0.txt DESTINATION share/spellcross-converter)
  install(FILES vendor/reloaded/docs/legal/NOTICE.md DESTINATION share/spellcross-converter)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(converter_tests tests/inventory_tests.cpp)
  target_link_libraries(converter_tests PRIVATE spellcross_converter_core)
  add_test(NAME inventory COMMAND converter_tests)
  if(SC_CONVERTER_GUI)
    add_executable(pak_io_tests tests/pak_io_tests.cpp)
    target_link_libraries(pak_io_tests PRIVATE spellcross_converter_core ${wxWidgets_LIBRARIES})
    add_test(NAME pak_io COMMAND pak_io_tests)
  endif()
  find_package(Python3 QUIET COMPONENTS Interpreter)
  if(SC_BUNDLED_FFMPEG)
    set(SC_FFMPEG "$<TARGET_FILE_DIR:SpellcrossConverter>/ffmpeg.exe")
    add_test(NAME bundled_ffmpeg COMMAND "${SC_FFMPEG}" -version)
  else()
    find_program(SC_FFMPEG ffmpeg)
  endif()
  if(Python3_Interpreter_FOUND AND SC_FFMPEG)
    add_test(NAME ffmpeg_lossless COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_ffmpeg_roundtrip.py" "${SC_FFMPEG}")
  endif()
endif()
